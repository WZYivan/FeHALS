import numpy as np
from typing import Dict, List, Tuple, Optional
import logging

logger = logging.getLogger(__name__)

class CoverageAnalyzer:
    """点云覆盖度分析器"""
    
    def __init__(self, grid_size: int = 50):
        """
        Args:
            grid_size: 网格分辨率（grid_size x grid_size）
        """
        self.grid_size = grid_size
    
    def analyze(self, points: np.ndarray) -> Dict:
        """
        分析点云覆盖度
        
        Args:
            points: (N, 3) 点云数组，XYZ坐标
            
        Returns:
            {
                'grid': (grid_size, grid_size) 密度矩阵,
                'bounds': [xmin, xmax, ymin, ymax],
                'statistics': {
                    'total_points': int,
                    'max_density': float,
                    'mean_density': float,
                    'coverage_percentage': float
                }
            }
        """
        if len(points) == 0:
            return self._empty_result()
        
        # 提取XY坐标
        xy = points[:, :2]
        xmin, xmax = xy[:, 0].min(), xy[:, 0].max()
        ymin, ymax = xy[:, 1].min(), xy[:, 1].max()
        
        # 防止边界重叠
        if xmax - xmin < 1e-6:
            xmin, xmax = xmin - 0.5, xmax + 0.5
        if ymax - ymin < 1e-6:
            ymin, ymax = ymin - 0.5, ymax + 0.5
        
        # 创建网格
        grid_x = np.linspace(xmin, xmax, self.grid_size)
        grid_y = np.linspace(ymin, ymax, self.grid_size)
        grid_xv, grid_yv = np.meshgrid(grid_x, grid_y, indexing='ij')
        
        # 计算每个网格单元的点数（使用直方图）
        hist, _, _ = np.histogram2d(
            xy[:, 0], xy[:, 1],
            bins=[self.grid_size, self.grid_size],
            range=[[xmin, xmax], [ymin, ymax]]
        )
        
        # 计算统计信息
        total_points = len(points)
        max_density = hist.max()
        mean_density = hist.mean()
        
        # 覆盖度百分比（密度 > 0 的网格占比）
        coverage_percentage = (hist > 0).sum() / (self.grid_size * self.grid_size) * 100
        
        return {
            'grid': hist.tolist(),
            'bounds': [float(xmin), float(xmax), float(ymin), float(ymax)],
            'statistics': {
                'total_points': total_points,
                'max_density': int(max_density),
                'mean_density': float(mean_density),
                'coverage_percentage': float(coverage_percentage)
            }
        }
    
    def _empty_result(self) -> Dict:
        """空点云结果"""
        return {
            'grid': [[0] * self.grid_size for _ in range(self.grid_size)],
            'bounds': [0, 1, 0, 1],
            'statistics': {
                'total_points': 0,
                'max_density': 0,
                'mean_density': 0,
                'coverage_percentage': 0
            }
        }

    def generate_heatmap_colors(self, density_matrix: np.ndarray) -> np.ndarray:
        """
        生成热力图颜色矩阵（用于前端渲染）
        
        Returns:
            (grid_size, grid_size, 4) RGBA数组
        """
        if density_matrix.max() == 0:
            return np.zeros((self.grid_size, self.grid_size, 4), dtype=np.uint8)
        
        # 归一化密度（0-1）
        normalized = density_matrix / density_matrix.max()
        
        # 颜色映射：蓝 -> 青 -> 黄 -> 红
        colors = np.zeros((self.grid_size, self.grid_size, 4), dtype=np.uint8)
        
        # 使用jet colormap的简化版本
        for i in range(self.grid_size):
            for j in range(self.grid_size):
                val = normalized[i, j]
                if val == 0:
                    colors[i, j] = [0, 0, 0, 0]  # 透明
                elif val < 0.5:
                    # 蓝到青
                    t = val / 0.5
                    colors[i, j] = [
                        0,
                        int(255 * t),
                        int(255),
                        200  # 半透明
                    ]
                else:
                    # 青到黄到红
                    t = (val - 0.5) / 0.5
                    colors[i, j] = [
                        int(255 * t),
                        int(255 * (1 - t)),
                        0,
                        200
                    ]
        
        return colors
